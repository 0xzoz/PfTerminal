class Wire:
    def __init__(self, proc):
        self.proc, self.raw, self.err, self.buffer = proc, bytearray(), bytearray(), bytearray()
        self.reads = selectors.DefaultSelector()
        for stream, kind in ((proc.stdout, 'out'), (proc.stderr, 'err')):
            os.set_blocking(stream.fileno(), False)
            self.reads.register(stream, selectors.EVENT_READ, kind)
        os.set_blocking(proc.stdin.fileno(), False)

    def exchange(self, request):
        raw = json.dumps(request, allow_nan=False).encode() + b'\n'
        if len(raw) > 8192:
            raise ValueError('request too long')
        save(run / f'request-{request["seq"]}.json', request)
        deadline = time.monotonic() + 10
        pending = memoryview(raw)
        while pending:
            if time.monotonic() > deadline:
                raise TimeoutError('stdin')
            try:
                n = os.write(self.proc.stdin.fileno(), pending)
                if not n:
                    raise EOFError('stdin closed')
                pending = pending[n:]
            except BlockingIOError:
                import select
                select.select([], [self.proc.stdin], [], 0.05)
        while b'\n' not in self.buffer:
            if time.monotonic() > deadline:
                raise TimeoutError('guest response')
            ready = self.reads.select(0.05)
            if not self.reads.get_map():
                raise EOFError('guest ended before response')
            for key, _ in ready:
                data = os.read(key.fileobj.fileno(), 65536)
                if not data:
                    self.reads.unregister(key.fileobj)
                    continue
                if key.data == 'err':
                    self.err.extend(data)
                    if len(self.err) > 262144:
                        raise ValueError('stderr budget')
                else:
                    self.raw.extend(data)
                    self.buffer.extend(data)
                    if len(self.raw) > 32 * 1024 * 1024 or len(self.buffer) > 3 * 1024 * 1024:
                        raise ValueError('observation budget')
        raw, _, self.buffer = self.buffer.partition(b'\n')
        response = guest.strict_json(raw)
        save(run / f'response-{request["seq"]}.json', response)
        if not isinstance(response, dict) or response.get('run_id') != run.name or response.get('case_id') != 'DEC-001':
            raise ValueError('response identity')
        if request['op'] != 'finish' and (response.get('seq') != request['seq'] or response.get('status') == 'terminal'):
            raise ValueError('failed or mismatched response')
        return response
