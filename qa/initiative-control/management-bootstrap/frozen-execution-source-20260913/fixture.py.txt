def publish_area(area, manager):
    assert subprocess.check_output(['git','-C',str(SOURCE),'rev-parse','HEAD'],text=True).strip() == 'c77123a7e9d4540a5b3ff19a5a9dc0f12848c923'
    assert not subprocess.check_output(['git','-C',str(SOURCE),'status','--porcelain'],text=True).strip()
    install = area / 'corbanu-control'
    number = len(list((install/'incoming').glob('upload-*'))) + 1
    target = install / 'incoming' / ('upload-' + format(number,'032x'))
    manifest = export.export(SOURCE, manager, target)
    pin = ROOT/'export-source-pin.json'
    if pin.exists():
        assert manifest['files'] == json.loads(pin.read_text())
    else:
        control.atomic_json(pin, manifest['files'])
    with patch.object(activate.subprocess, 'run') as services:
        activate.activate(install, target, area / 'units-not-installed')
        control.atomic_json(area / 'stubbed-services.json', [list(c.args[0]) for c in services.call_args_list])
    # Run the real publisher from the exact exported generation, using existing Python.
    cmd = [sys.executable,'-B',str(target/'source/scripts/initiative_control/control.py'),'publish',
        '--repo',str(target/'source'),'--state',str(install/'state'),'--output',str(install/'site')]
    started = stamp()
    result = subprocess.run(cmd, capture_output=True,text=True)
    control.atomic_json(area / f'publication-{number}.json', {'started_at':started,'finished_at':stamp(),'command':cmd,'returncode':result.returncode,'stdout':result.stdout,'stderr':result.stderr})
    if result.returncode: raise RuntimeError(result.stderr)
    shutil.copytree(install/'site/current', area/f'retained-publication-{number}')
    assert all(hashlib.sha256((SOURCE/p).read_bytes()).hexdigest()==h for p,h in manifest['files'].items())
    print(area.name, manifest['decision_feed']['status'], manifest['decision_feed']['assessed_at'], flush=True)

def advance(name, state):
    area = ROOT/'fixtures'/name
    manager = area/'manager'
    old = decisions.load_fixture(manager, stamp())
    value = copy.deepcopy(old) if old else base(stamp())
    value['revision'] = old['revision']+1 if old else 1
    value['assessed_at'] = stamp()
    if state in ('acknowledged','resolved'):
        record = copy.deepcopy(value['decisions'][0]['revisions'][-1])
        record['revision'] += 1
        record['status'] = state
        record['updated_at'] = stamp()
        if state == 'resolved':
            record['resolution'] = {'answered_revision':1,'actor':'Avery','answer':'Five testers','scope':'Synthetic pilot only','recorded_at':stamp()}
        value['decisions'][0]['revisions'].append(record)
    elif state == 'newer-mixed':
        r=value['decisions'][0]['revisions'][0]
        r.update(owner=None,impact=None,raised_at=stamp(-1600),updated_at=stamp(-1400))
        r['evidence'][0]['assessed_at']=stamp(-1500)
        r['evidence'].append({'label':'Unassessed sanitized evidence','path':'qa/unpublished-synthetic.md','assessed_at':None})
    decisions.save_fixture(manager,value,decisions.digest(old) if old else None,stamp())
    publish_area(area,manager)

def select(name):
    target = ROOT / 'fixtures' / name / 'corbanu-control/site'
    if not (target/'current/index.html').is_file(): raise ValueError('not published')
    pointer = ROOT / 'serving-next'
    pointer.symlink_to(target)
    os.replace(pointer, ROOT / 'serving')
    print('Selected', name)

def serve():
    original = control.ThreadingHTTPServer
    def bind(address, handler):
        server = original(address, handler)
        control.atomic_json(ROOT/'server.json', {'pid':os.getpid(),'host':server.server_address[0],'port':server.server_address[1],'url':f'http://127.0.0.1:{server.server_address[1]}/index.html'})
        print('LOOPBACK', server.server_address, flush=True)
        return server
    with patch.object(control, 'ThreadingHTTPServer', bind):
        control.serve(ROOT/'serving', 0)

if __name__ == '__main__':
    os.umask(0o077)

        for name in sys.argv[2:]: build(name)
    elif sys.argv[1] == 'select': select(sys.argv[2])
    elif sys.argv[1] == 'serve': serve()
    elif sys.argv[1] == 'advance': advance(sys.argv[2],sys.argv[3])
